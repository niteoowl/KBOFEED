const renderGames = (data) => {
  const games = data.result?.games || []; // .gameList가 아니라 .games 임을 명시
  if (!games.length) return console.log("경기가 없습니다.");
  
  games.forEach(game => {
    console.log(`${game.homeTeamName} vs ${game.awayTeamName} (${game.statusInfo})`);
    // 이미지 처리 시 반드시 EmblemUrl 필드 사용 (404 방지)
    const homeLogo = game.homeTeamEmblemUrl; 
  });
};

// 2. 순위표 렌더링 시
const renderRankings = (data) => {
  const rankings = data.result?.seasonTeamStats || []; // .teamList가 아니라 .seasonTeamStats 임을 명시
  rankings.map(team => {
    console.log(`${team.ranking}위: ${team.teamName} (승률: ${team.wra})`);
  });
};

// 3. 실시간 주자 정보 렌더링 시
const renderRelay = (data) => {
  const state = data.result?.currentGameState;
  if (state) {
    const { base1, base2, base3, strike, ball, out } = state;
    console.log(`주자 상황: ${base1}-${base2}-${base3} | ${ball}B ${strike}S ${out}O`);
  }
};
